    closing_time TIME,
    one_c_endpoint TEXT,
    is_active BOOLEAN DEFAULT true
);

-- 3. Базовый справочник медикаментов (МНН, аналоги)
CREATE TABLE medicines (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    trade_name VARCHAR(255) NOT NULL,
    inn_name VARCHAR(255) NOT NULL, -- Международное непатентованное название
    barcode VARCHAR(64) UNIQUE,
    category_id INT NOT NULL,
    dosage_form VARCHAR(100) NOT NULL, -- таблетки, сироп, ампулы
    dosage_strength VARCHAR(100) NOT NULL, -- 500 мг, 10 мг/мл
    manufacturer_country VARCHAR(100) NOT NULL,
    manufacturer_name VARCHAR(255) NOT NULL,
    is_prescription_required BOOLEAN DEFAULT false,
    storage_temperature VARCHAR(50),
    description_tj TEXT,
    description_ru TEXT,
    image_url TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 4. Остатки и цены в конкретных аптеках (Синхронизация с 1С)
CREATE TABLE pharmacy_inventory (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    stock_quantity INT NOT NULL DEFAULT 0,
    batch_number VARCHAR(100),
    expiry_date DATE NOT NULL,
    last_synced_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    CONSTRAINT unique_pharmacy_medicine UNIQUE (pharmacy_id, medicine_id)

CREATE TABLE orders (

    delivery_longitude NUMERIC(11, 8),

    prescription_image_url TEXT,

-- 6. Детализация позиций заказа
CREATE TABLE order_items (

    total_price_tjs NUMERIC(10, 2) NOT NULL

3. Детальная спецификация ключевых модулей
МОДУЛЬ 1: Умный поиск медикаментов и подбор аналогов (Smart Search & MNN Engine)
Проблема: Пользователь ищет "Мезим", которого нет в ближайшей аптеке, или хочет сэкономить.
Логика работы сервиса:
Поисковый запрос проходит через полнотекстовый поиск Elasticsearch с нечетким соответствием (Fuzzy match / расстояние Левенштейна 
 для учета ошибок правописания: «цытрамон» 
 «цитрамон»).
Если препарат найден: выдается список аптек, отсортированный по:
Блок аналогов по МНН: Сервис автоматически выполняет вторичный запрос:
code
SQL
SELECT m.*, pi.price_tjs, p.name as pharmacy_name 
FROM medicines m
JOIN pharmacy_inventory pi ON m.id = pi.medicine_id
JOIN pharmacies p ON pi.pharmacy_id = p.id
WHERE m.inn_name = (SELECT inn_name FROM medicines WHERE id = :current_med_id)
  AND m.id != :current_med_id
  AND pi.stock_quantity > 0
ORDER BY pi.price_tjs ASC;
Пользователь видит плашку: «Сэкономьте до 65 сомони: найден дженерик с тем же действующим веществом (Панкреатин 10,000 ЕД) за 18 сомони вместо 83 сомони».
МОДУЛЬ 2: OCR-распознавание рецептов и голосовых сообщений (AI Medical Input)
Входные данные:
Фотография рецепта врача, сделанная на камеру смартфона.
Голосовое сообщение на таджикском/русском языке (например: «Салом алейкум, ба ман барои фишори баланд каптоприл ва аспирин кардио лозим»).
Пайплайн обработки:
Аудио: Запись передается в Whisper API (дообученный на фарм-глоссарии и таджикских диалектах) 
 генерация текстового промпта.
Фото: Изображение нормализуется, бинаризуется и отправляется в мультимодальную модель Gemini 2.5 Flash со строгим Structured Output:
